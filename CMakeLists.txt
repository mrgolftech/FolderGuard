cmake_minimum_required(VERSION 3.20)
project(FolderGuard VERSION 3.0.0 LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "FolderGuard is a Windows-only utility")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(FolderGuard WIN32 src/Main.cpp src/Security.cpp src/Security.h src/FolderGuard.rc)
target_compile_definitions(FolderGuard PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX WINVER=0x0601 _WIN32_WINNT=0x0601)
target_compile_options(FolderGuard PRIVATE $<$<CXX_COMPILER_ID:MSVC>:/utf-8> $<$<CXX_COMPILER_ID:MSVC>:/W4>)
target_link_libraries(FolderGuard PRIVATE d2d1 dwrite ole32 shell32 advapi32 user32 gdi32)
# Static VC runtime: don't require MSVC redistributable installation on older Windows.
if(MSVC)
  set_property(TARGET FolderGuard PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  target_link_options(FolderGuard PRIVATE /MANIFEST:NO)
endif()
set_target_properties(FolderGuard PROPERTIES OUTPUT_NAME FolderGuard)

# Smoke tests on temporary NTFS directory; no real user files are touched.
enable_testing()
add_executable(FolderGuardSmoke tests/SecuritySmoke.cpp src/Security.cpp src/Security.h)
target_include_directories(FolderGuardSmoke PRIVATE src)
target_compile_definitions(FolderGuardSmoke PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX WINVER=0x0601 _WIN32_WINNT=0x0601)
target_compile_options(FolderGuardSmoke PRIVATE $<$<CXX_COMPILER_ID:MSVC>:/utf-8> $<$<CXX_COMPILER_ID:MSVC>:/W4>)
target_link_libraries(FolderGuardSmoke PRIVATE advapi32 shell32)
if(MSVC)
  set_property(TARGET FolderGuardSmoke PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
add_test(NAME SecuritySmoke COMMAND FolderGuardSmoke)
